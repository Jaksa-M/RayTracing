#version 430 core
// Copies the rendered image (CPU or GPU path) to the screen framebuffer.
// Values in the texture are linear, GL_FRAMEBUFFER_SRGB does the gamma conversion on write.
in vec2 v_uv;
out vec4 FragColor;

uniform sampler2D u_screen_tex;

void main() {
    FragColor = vec4(texture(u_screen_tex, v_uv).rgb, 1.0);
}
